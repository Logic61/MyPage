    1. Python 实现

    使用 NumPy 可以非常方便地完成：

    import numpy as np

# 均值向量
mu = np.array([0.0, 0.0])

# 协方差矩阵
Sigma = np.array([
    [1.0, 0.6],
    [0.6, 1.0]
])

# Cholesky 分解
L = np.linalg.cholesky(Sigma)

# 生成 100000 个独立标准正态向量
rng = np.random.default_rng(42)
Z = rng.standard_normal((100000, 2))

# 线性变换
X = Z @ L.T + mu

# 检查经验均值和协方差
print(X.mean(axis=0))
print(np.cov(X, rowvar=False))

    NumPy 实现：

    import numpy as np

def gaussian_logpdf(x, mu, Sigma):
    L = np.linalg.cholesky(Sigma)

    # 解 Lr = x - mu
    r = np.linalg.solve(L, x - mu)

    # log(det(Sigma))
    log_det = 2 * np.log(np.diag(L)).sum()

    n = len(mu)

    # 对数密度
    return -0.5 * (
        n * np.log(2 * np.pi)
        + log_det
        + r @ r
    )

mu = np.array([0.0, 0.0])
Sigma = np.array([
    [1.0, 0.6],
    [0.6, 1.0]
])

x = np.array([1.0, 0.0])

log_p = gaussian_logpdf(x, mu, Sigma)
density = np.exp(log_p)

print(log_p)
print(density)

    实际机器学习中，对数密度通常比直接计算密度更有用，因为多个概率密度相乘容易造成数值下溢。
